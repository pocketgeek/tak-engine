cmake_minimum_required(VERSION 3.20)
# Packaging only: never relink, strip or otherwise mutate signed executables.
if(NOT TAK_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "TAK_VERSION must be a release version")
endif()
project(tak-engine VERSION ${TAK_VERSION} LANGUAGES NONE)
# This installer carries x64 binaries; LANGUAGES NONE does not detect pointer size.
set(CMAKE_SIZEOF_VOID_P 8)
if(NOT IS_DIRECTORY "${TAK_SIGNED_PAYLOAD}")
  message(FATAL_ERROR "TAK_SIGNED_PAYLOAD must point at the verified x64 release payload")
endif()
foreach(name takclient.exe takserver.exe cartographer.exe crusades_admin.exe
             STREAMING-LICENSES.txt EDITOR-FONT-LICENSE.txt ACME-LICENSES.txt)
  if(NOT EXISTS "${TAK_SIGNED_PAYLOAD}/${name}")
    message(FATAL_ERROR "Missing installer payload: ${name}")
  endif()
  install(FILES "${TAK_SIGNED_PAYLOAD}/${name}" DESTINATION .)
endforeach()
set(TAK_PACKAGE_CARTOGRAPHER ON)
include(installer.cmake)
include(CPack)
